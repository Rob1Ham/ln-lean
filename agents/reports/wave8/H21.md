### Stored payload model (additive, local)

```lean
structure AcceptStore where
  handshake : HandshakeState := HandshakeState.initial
  acceptChannel : Option AcceptChannel := none

inductive StoreEvent where
  | handshake (e : HandshakeEvent)
  | recvAccept (a : AcceptChannel)

def storeStep (st : AcceptStore) : StoreEvent → Option AcceptStore
inductive StoreReachable : AcceptStore → Prop
def AcceptStore.PayloadFlagInv (st : AcceptStore) : Prop :=
  (st.handshake.acceptReceived = true) ↔ (st.acceptChannel.isSome = true)
```

```lean
theorem storeStep_recvAccept_state {st st' : AcceptStore} {a : AcceptChannel}
    (h : storeStep st (.recvAccept a) = some st') :
    ∃ s' : HandshakeState,
      handshakeStep st.handshake .recvAcceptChannel = some s' ∧
        st.acceptChannel = none ∧ st'.handshake = s' ∧
        st'.acceptChannel = some a

theorem storeStep_recvAccept_sets_payload {st st' : AcceptStore}
    {a : AcceptChannel} (h : storeStep st (.recvAccept a) = some st') :
    st'.acceptChannel = some a

theorem storeStep_recvAccept_sets_flag {st st' : AcceptStore}
    {a : AcceptChannel} (h : storeStep st (.recvAccept a) = some st') :
    st'.handshake.acceptReceived = true

theorem storeStep_payload_preserved {st st' : AcceptStore} {e : StoreEvent}
    {a : AcceptChannel} (h : storeStep st e = some st')
    (hp : st.acceptChannel = some a) : st'.acceptChannel = some a

theorem storeReachable_payloadFlagInv {st : AcceptStore}
    (hr : StoreReachable st) : st.PayloadFlagInv

theorem stored_payload_iff_acceptReceived {st : AcceptStore}
    (hr : StoreReachable st) :
    st.acceptChannel.isSome = true ↔ st.handshake.acceptReceived = true

theorem receivedStore_reachable (a : AcceptChannel) :
    StoreReachable (receivedStore a)
```

The bare `recvAcceptChannel` event is disabled at this layer
(`storeStep_handshake_recvAcceptChannel`), so a store cannot set
`acceptReceived` without the payload.

### Transport attribution and stored-key registration

```lean
structure ReceivedAccept where
  sender : PeerIdentity
  message : AcceptChannel

def AcceptAttributed (sender : PeerIdentity) (a : AcceptChannel) : Prop :=
  sender.key = a.fundingPubKey

def storedAcceptNegotiation (st : AcceptStore) (localFundingKey : PubKey)
    (fundingSatoshis : BtcAmount) (pushMsat : Msat)
    (temporaryChannelId : TemporaryChannelId) :
    Option LnLean.Bitcoin.FundingNegotiation

theorem stored_accept_key_registration {M : PrimitiveModel} (K : KeyModel M)
    (st : AcceptStore) (a : AcceptChannel) (seed : ByteArray)
    (sender : PeerIdentity) (hstore : st.acceptChannel = some a)
    (hadv : AdvertisesKey K seed a) (hattr : AcceptAttributed sender a) :
    st.acceptChannel.isSome = true ∧
      a.fundingPubKey = K.publicOfSeed seed ∧
      sender.key = K.publicOfSeed seed ∧
      sender.key = a.fundingPubKey ∧
      (∀ (localFundingKey : PubKey) (fundingSatoshis : BtcAmount)
        (pushMsat : Msat) (temporaryChannelId : TemporaryChannelId),
        (a.toFundingNegotiation localFundingKey fundingSatoshis pushMsat
          temporaryChannelId).remoteFundingKey = K.publicOfSeed seed)

theorem stored_accept_funding_signature_valid {M : PrimitiveModel}
    (K : KeyModel M) (st : AcceptStore) (a : AcceptChannel) (seed : ByteArray)
    (sender : PeerIdentity) (_hstore : st.acceptChannel = some a)
    (hadv : AdvertisesKey K seed a) (hattr : AcceptAttributed sender a)
    (fundingTx : ByteArray) (hdom : K.correctOn (fundingTag ++ fundingTx)) :
    M.ecdsaVerify a.fundingPubKey (fundingTag ++ fundingTx)
        (K.sign (K.secretOfSeed seed) (fundingTag ++ fundingTx)) = true ∧
      M.ecdsaVerify sender.key (fundingTag ++ fundingTx)
        (K.sign (K.secretOfSeed seed) (fundingTag ++ fundingTx)) = true

theorem stored_accept_signature_attribution {M : PrimitiveModel}
    (K : KeyModel M) (hkb : KeyBinding M.ecdsaVerify) (st : AcceptStore)
    (a : AcceptChannel) (seed : ByteArray) (sender : PeerIdentity)
    (_hstore : st.acceptChannel = some a) (hadv : AdvertisesKey K seed a)
    (hattr : AcceptAttributed sender a) (fundingTx : ByteArray)
    (hdom : K.correctOn (fundingTag ++ fundingTx)) {pk' : PubKey}
    (hver : M.ecdsaVerify pk' (fundingTag ++ fundingTx)
      (K.sign (K.secretOfSeed seed) (fundingTag ++ fundingTx)) = true) :
    pk' = sender.key

theorem stored_accept_funding_signed_valid {M : PrimitiveModel}
    (K : KeyModel M) (hwire : WireShaped K) (st : AcceptStore)
    (a : AcceptChannel) (seed : ByteArray) (sender : PeerIdentity)
    (_hstore : st.acceptChannel = some a) (hadv : AdvertisesKey K seed a)
    (hattr : AcceptAttributed sender a) (channelId : ChannelId)
    (fundingTx : ByteArray) (hdom : K.correctOn (fundingTag ++ fundingTx)) :
    fundingSignedValid M sender.key fundingTx
      { channelId := channelId
        signature :=
          K.sign (K.secretOfSeed seed) (fundingTag ++ fundingTx) }
```

### Proof-of-possession premise (explicit) and its sharp limit

```lean
def registrationTag : ByteArray := ByteArray.mk #[0x72, 0x65, 0x67]
def registrationChallenge (a : AcceptChannel) : ByteArray :=
  registrationTag ++ a.fundingPubKey.bytes

structure PossessionProof (M : PrimitiveModel) (a : AcceptChannel) where
  responder : PubKey
  signature : EcdsaSig
  verifies : M.ecdsaVerify a.fundingPubKey (registrationChallenge a) signature = true

theorem possession_binds_responder {M : PrimitiveModel}
    (hkb : KeyBinding M.ecdsaVerify) {a : AcceptChannel}
    (p : PossessionProof M a)
    (hresp : M.ecdsaVerify p.responder (registrationChallenge a) p.signature = true) :
    p.responder = a.fundingPubKey

theorem possession_gives_transport_attribution {M : PrimitiveModel}
    (K : KeyModel M) (hkb : KeyBinding M.ecdsaVerify) (st : AcceptStore)
    (a : AcceptChannel) (seed : ByteArray) (sender : PeerIdentity)
    (_hstore : st.acceptChannel = some a) (_hadv : AdvertisesKey K seed a)
    (p : PossessionProof M a) (hresponder : p.responder = sender.key)
    (hresp : M.ecdsaVerify p.responder (registrationChallenge a) p.signature = true) :
    AcceptAttributed sender a

theorem stored_accept_registration_of_possession {M : PrimitiveModel}
    (K : KeyModel M) (hkb : KeyBinding M.ecdsaVerify) (st : AcceptStore)
    (a : AcceptChannel) (seed : ByteArray) (sender : PeerIdentity)
    (hstore : st.acceptChannel = some a) (hadv : AdvertisesKey K seed a)
    (p : PossessionProof M a) (hresponder : p.responder = sender.key)
    (hresp : M.ecdsaVerify p.responder (registrationChallenge a) p.signature = true) :
    sender.key = K.publicOfSeed seed ∧ sender.key = a.fundingPubKey
```

### Retained negatives and non-vacuity

```lean
theorem possession_not_attribution_without_keyBinding (M₀ : PrimitiveModel) :
    ∃ (M : PrimitiveModel) (a : AcceptChannel) (p : PossessionProof M a),
      p.responder ≠ a.fundingPubKey ∧
        M.ecdsaVerify p.responder (registrationChallenge a) p.signature = true ∧
        EufCma M.ecdsaVerify ∧
        ¬ KeyBinding M.ecdsaVerify

theorem same_accept_payload_distinct_senders :
    ∃ r₁ r₂ : ReceivedAccept,
      r₁.message = r₂.message ∧ r₁.sender ≠ r₂.sender

theorem stored_accept_payload_not_signer_attribution (M₀ : PrimitiveModel) :
    ∃ (M : PrimitiveModel) (K : KeyModel M) (st : AcceptStore)
      (a : AcceptChannel) (seed₁ seed₂ : ByteArray) (fundingTx : ByteArray),
      StoreReachable st ∧
        st.acceptChannel = some a ∧
        AdvertisesKey K seed₁ a ∧
        K.publicOfSeed seed₁ ≠ K.publicOfSeed seed₂ ∧
        EufCma M.ecdsaVerify ∧
        M.ecdsaVerify a.fundingPubKey (fundingTag ++ fundingTx)
          (K.sign (K.secretOfSeed seed₂) (fundingTag ++ fundingTx)) = true ∧
        ¬ KeyBinding M.ecdsaVerify

theorem stored_accept_registration_nonvacuous (M₀ : PrimitiveModel) (k₀ : PubKey) :
    ∃ (M : PrimitiveModel) (K : KeyModel M) (st : AcceptStore)
      (a : AcceptChannel) (seed : ByteArray) (sender : PeerIdentity)
      (fundingTx : ByteArray),
      StoreReachable st ∧ st.acceptChannel = some a ∧
        st.handshake.acceptReceived = true ∧
        EufCma M.ecdsaVerify ∧ KeyBinding M.ecdsaVerify ∧
        AdvertisesKey K seed a ∧ AcceptAttributed sender a ∧
        K.correctOn (fundingTag ++ fundingTx) ∧
        M.ecdsaVerify sender.key (fundingTag ++ fundingTx)
          (K.sign (K.secretOfSeed seed) (fundingTag ++ fundingTx)) = true

theorem possession_proof_nonvacuous (M₀ : PrimitiveModel) (k₀ : PubKey) :
    ∃ (M : PrimitiveModel) (a : AcceptChannel) (p : PossessionProof M a),
      KeyBinding M.ecdsaVerify ∧ p.responder = a.fundingPubKey

theorem stored_accept_registration_summary :
    -- (1) stored payload + AdvertisesKey + transport attribution ⇒ sender.key is
    --     the signer's generated key, and the payload is present;
    -- (2) KeyBinding ⇒ every accepting key equals sender.key;
    -- (3) PoP + KeyBinding ⇒ transport attribution;
    -- (4) ∃ one payload, two transport senders;
    -- (5) ∀ base model, ∃ reachable store with payload + AdvertisesKey + C3
    --     where a different secret key's signature verifies and KeyBinding fails.
```

## Scope caveats

- **`AdvertisesKey` is still a hypothesis**, not derived here.  The module
  proves exactly what follows from it once a payload is stored; it does not
  prove that the payload's `funding_pubkey` was honestly generated by the peer.
- **Transport attribution is explicit.**  `AcceptAttributed sender a` is
  `sender.key = a.fundingPubKey`, the same shape as
  `FundingAuthTargets.OpenChannelAttributed`.  `same_accept_payload_distinct_senders`
  shows the payload alone cannot establish it; ASSUMPTIONS §3/§6 still applies
  (Noise is modelled as an ideal authenticated channel with a peer identity).
- **Proof of possession is only a transcript model.**  `PossessionProof` has no
  freshness, challenge sampling, adversary, or game; `registrationChallenge` is
  a domain-tagged key encoding, not a nonce exchange.  It becomes meaningful
  only under `KeyBinding`; `possession_not_attribution_without_keyBinding`
  mechanises that without `KeyBinding` it proves nothing.
- **`KeyBinding` is not derived from C3.**  It is the unchanged wave-E
  contract, and `stored_accept_payload_not_signer_attribution` lifts the wave-7
  `key_generation_not_key_attribution` counterexample to a reachable store.
  The store records which key was advertised, never who signed.
- **No EUF-CMA strengthening is assumed.**  `EufCma` appears only as the
  existing `PrimitiveModel.ecdsa_eufCma` field in the negative theorems.
- **The storage model is local and additive.**  `AcceptStore`, `StoreEvent`,
  `storeStep`, and `StoreReachable` are new names in this module; the shared
  `HandshakeState`, `HandshakeEvent`, and `handshakeStep` are untouched.  This
  overlaps lane H20's `acceptReceived?` modeling and must be reconciled by the
  parent (H21's model is designed to be replaced by the shared-field patch).
- **No wire decoding is claimed.**  `AcceptChannel` is the existing record;
  type-33 BOLT 2 decoding is out of scope.
- **`stored_accept_funding_signed_valid` uses `WireShaped`** (a separate
  hypothesis) for the BOLT 1 wire shape; no signature encoding is synthesized.

## Shared-file integration patch (parent-owned; not applied)

The gap is that `HandshakeState` has no payload.  A non-breaking additive patch
is:

```lean
-- LnLean/Basic/Core.lean (additive instances; no existing DecidableEq)
structure PubKey where
  bytes : ByteArray
deriving Repr, BEq, DecidableEq

structure ChannelId where
  bytes : ByteArray
deriving Repr, BEq, DecidableEq

-- LnLean/Protocol/Messages.lean (additive instance)
structure AcceptChannel where
  ...
deriving Repr, BEq, DecidableEq

-- LnLean/Protocol/Handshake.lean
structure HandshakeState where
  ...
  acceptChannel : Option AcceptChannel := none
deriving Repr, DecidableEq, BEq

/-- Payload-carrying `accept_channel` receipt.  It reuses the exact
`handshakeStep` guard and additionally stores the message. -/
def recvAcceptChannelWith (s : HandshakeState) (a : AcceptChannel) :
    Option HandshakeState :=
  if s.canRecvAccept then
    some { s with acceptReceived := true, acceptChannel := some a }
  else none

theorem recvAcceptChannelWith_state (s s' : HandshakeState)
    (a : AcceptChannel) (h : recvAcceptChannelWith s a = some s') :
    s'.acceptReceived = true ∧ s'.acceptChannel = some a := by
  by_cases hg : s.canRecvAccept
  · simp [recvAcceptChannelWith, hg] at h
    exact ⟨by rw [← h], by rw [← h]⟩
  · simp [recvAcceptChannelWith, hg] at h

/-- The payload receipt agrees with the existing boolean milestone transition
on the ten-milestone state. -/
theorem recvAcceptChannelWith_agrees (s s' : HandshakeState)
    (a : AcceptChannel) (h : recvAcceptChannelWith s a = some s') :
    handshakeStep s .recvAcceptChannel =
      some { s' with acceptChannel := s.acceptChannel } := by
  by_cases hg : s.canRecvAccept
  · simp [recvAcceptChannelWith, handshakeStep, hg] at h ⊢
    rw [← h]
  · simp [recvAcceptChannelWith, hg] at h
```

Integration sketch for the parent: replace `AcceptStore.handshake` by
`HandshakeState`, replace `storeStep`'s `.recvAccept a` branch by
`recvAcceptChannelWith`, and keep every theorem in §2–§5 unchanged (they only
speak about `acceptChannel`, `acceptReceived`, and the stored payload).  If the
parent prefers H20's `acceptReceived?` field, use that field as
`acceptChannel` and keep `stored_payload_iff_acceptReceived` as the bridge.
`AcceptChannel`'s new `DecidableEq` is additive; it should be verified in the
full gate because `HandshakeState`'s existing `DecidableEq` consumers are
outside this lane.
