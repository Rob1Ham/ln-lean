## Exact result

namespace LnLean.Proofs.WaveH.H11

-- Wire decomposition / parser inversion.
theorem toTx_eq (h : HtlcWire) :
    h.toTx = { version := h.version.toInt32
               vin := [htlcIn h]
               vout := [htlcOut h]
               locktime := h.locktime
               witness := [[]] }

theorem bL_txBytes_eq (h : HtlcWire) :
    bL h.txBytes =
      prefixBytes h ++ (encCSL h.scriptPubKey.size ++
        (bL h.scriptPubKey ++ encU32L h.locktime))

theorem decTxCore_prefix (h : HtlcWire) (htxid : h.commitmentTxid.size = 32)
    (tail : Bytes) :
    decTxCore (prefixBytes h ++ tail) =
      (do
        let (s, l1) ← decScriptL tail
        let (lt, l2) ← decU32L l1
        .ok (parsedWith h s lt, l2))

-- wireDomain is exactly WellFormed (no hidden premise).
theorem wellFormed_of_wireDomain (h : HtlcWire)
    (hw : Bitcoin.Serialization.wireDomain h.toTx) : h.WellFormed

theorem wireDomain_iff_wellFormed (h : HtlcWire) :
    Bitcoin.Serialization.wireDomain h.toTx ↔ h.WellFormed

theorem wellFormed_field_bounds (h : HtlcWire) (hw : h.WellFormed) :
    h.commitmentTxid.size = 32 ∧
      h.scriptPubKey.size ≤ Bitcoin.Serialization.MAX_SIZE ∧
      h.valueSat.toNat < 9223372036854775808

-- Positive round-trip at the production entry point.
theorem txBytes_parse_full (h : HtlcWire) (hw : h.WellFormed) :
    Bitcoin.Serialization.decTxFull (bL h.txBytes) = .ok h.toTx

theorem txBytes_roundtrip (h : HtlcWire) (hw : h.WellFormed) :
    Bitcoin.Serialization.decTxFull (bL h.txBytes) = .ok h.toTx ∧
      bL h.txBytes = Bitcoin.Serialization.encTxL h.toTx

-- Amount characterization: parser returns toTx iff the value is int64-representable.
theorem u64ToAmt_eq (u : Nat) :
    u64ToAmt u = (u : Int) ∨
      u64ToAmt u = (u : Int) - 18446744073709551616

theorem u64ToAmt_toNat_eq_iff (v : UInt64) :
    u64ToAmt v.toNat = (v.toNat : Int) ↔ v.toNat < 9223372036854775808

theorem txBytes_parse_form (h : HtlcWire) (htxid : h.commitmentTxid.size = 32)
    (hscr : h.scriptPubKey.size ≤ Bitcoin.Serialization.MAX_SIZE) :
    Bitcoin.Serialization.decTxCore (bL h.txBytes) =
      .ok (parsedWith h ⟨h.scriptPubKey⟩ h.locktime, [])

theorem parse_eq_toTx_iff (h : HtlcWire) (htxid : h.commitmentTxid.size = 32)
    (hscr : h.scriptPubKey.size ≤ Bitcoin.Serialization.MAX_SIZE) :
    Bitcoin.Serialization.decTxCore (bL h.txBytes) = .ok (h.toTx, []) ↔
      h.valueSat.toNat < 9223372036854775808

theorem parse_ne_toTx_of_value_ge (h : HtlcWire)
    (htxid : h.commitmentTxid.size = 32)
    (hscr : h.scriptPubKey.size ≤ Bitcoin.Serialization.MAX_SIZE)
    (hv : 9223372036854775808 ≤ h.valueSat.toNat) :
    Bitcoin.Serialization.decTxCore (bL h.txBytes) ≠ .ok (h.toTx, [])

theorem parse_ne_toTx_of_value_ge_uint64 (h : HtlcWire)
    (htxid : h.commitmentTxid.size = 32)
    (hscr : h.scriptPubKey.size ≤ Bitcoin.Serialization.MAX_SIZE)
    (hv : (9223372036854775808 : UInt64) ≤ h.valueSat) :
    Bitcoin.Serialization.decTxCore (bL h.txBytes) ≠ .ok (h.toTx, [])

theorem not_wireDomain_of_value_ge (h : HtlcWire)
    (hv : (9223372036854775808 : UInt64) ≤ h.valueSat) :
    ¬ Bitcoin.Serialization.wireDomain h.toTx

theorem not_wireDomain_of_script_oversize (h : HtlcWire)
    (hsz : Bitcoin.Serialization.MAX_SIZE < h.scriptPubKey.size) :
    ¬ Bitcoin.Serialization.wireDomain h.toTx

-- Truncated inputs are hard `.truncated`.
theorem parse_truncated_at_version (h : HtlcWire) :
    Bitcoin.Serialization.decTxCore ((bL h.txBytes).take 4) = .error .truncated

theorem parse_truncated_in_script (h : HtlcWire)
    (htxid : h.commitmentTxid.size = 32) (hpos : 0 < h.scriptPubKey.size)
    (hM : h.scriptPubKey.size ≤ Bitcoin.Serialization.MAX_SIZE) :
    Bitcoin.Serialization.decTxCore
        ((bL h.txBytes).take ((prefixBytes h).length +
          compactSizeWidth h.scriptPubKey.size)) = .error .truncated

-- Declared length >= 2^32 is rejected by the 0xff branch.
theorem parse_script_tooLong (h : HtlcWire)
    (htxid : h.commitmentTxid.size = 32) :
    Bitcoin.Serialization.decTxCore (prefixBytes h ++ encCSL 4294967296) =
      .error .compactSizeTooLong

-- Honest domain gap: MAX_SIZE+1 script parses (0xfe branch has no MAX_SIZE check).
theorem decCS_encCSL_wide (n : Nat) (h1 : ¬ n < 0xfd) (h2 : ¬ n ≤ 0xffff)
    (h3 : n ≤ 0xffffffff) (rest : Bytes) :
    decCS (encCSL n ++ rest) = .ok (n, rest)

theorem oversizedWire_parse :
    Bitcoin.Serialization.decTxCore (bL oversizedWire.txBytes) =
      .ok (oversizedWire.toTx, [])

theorem oversizedWire_not_wellFormed : ¬ oversizedWire.WellFormed

theorem oversizedWire_not_wireDomain :
    ¬ Bitcoin.Serialization.wireDomain oversizedWire.toTx

end LnLean.Proofs.WaveH.H11

## Scope caveats

1. **`0xfe` `CompactSize` domain gap (the narrowed point).**
   `decCS` enforces `MAX_SIZE` only on the `0xff` branch. A declared script
   length `n` with `0xffff < n ≤ 0xffffffff` is accepted without the
   `MAX_SIZE` check. Hence scripts of size in
   `(Bitcoin.Serialization.MAX_SIZE, 2^32)` parse successfully but fail
   `wireDomain`/`WellFormed`; `oversizedWire_parse` is the concrete
   `MAX_SIZE + 1` witness. Only `n ≥ 2^32` is a parse error
   (`parse_script_tooLong`). Do not restate this lane as "oversized scripts are
   rejected".
2. **Amount out of range is semantic, not a parse error.**
   `decTxOutL` applies the signed two's-complement fold `u64ToAmt`; values
   `≥ 2^63` parse to a negative `Int` amount and differ from `h.toTx`.
   `parse_eq_toTx_iff` states the exact iff; `parse_ne_toTx_of_value_ge` is the
   negative. `WellFormed` (`valueSat < 2^63`) is the right guard.
3. **`MAX_SIZE` is the serialization bound.**
   Every occurrence is qualified as `Bitcoin.Serialization.MAX_SIZE`
   (`0x02000000 = 33554432`), not `Bitcoin.Data.MAX_SIZE` (`2^64 - 1`).
4. **Only the `HtlcWire` shape.** One input, one output, empty scriptSig,
   witness `[[]]`. No claim about arbitrary transactions, scriptCode
   synthesis, signature validity, consensus limits, or the BIP 143 digest
   (that is H09/H10/H12 territory).
5. `txBytes_parse_full` is the "no trailing bytes" `decTxFull` lift of the
   existing `HtlcWire.txBytes_parse`; the latter is not restated or modified.
