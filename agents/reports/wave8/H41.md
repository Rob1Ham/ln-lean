structure EpochNetwork where
  network : CanonicalNetwork
  epoch : Nat

noncomputable def epochAdmit (verify : SigVerifier) (es : EpochNetwork)
    (m : GossipMessage) : EpochNetwork
def epochAdvance (es : EpochNetwork) : EpochNetwork
def rotateNetwork (cs : CanonicalNetwork) (n : NodeId) (k : PubKey) : CanonicalNetwork
def epochRotate (es : EpochNetwork) (n : NodeId) (k : PubKey) : EpochNetwork

theorem epoch_admission_epoch_independent (verify : SigVerifier)
    (cs : CanonicalNetwork) (e₁ e₂ : Nat) (m : GossipMessage) :
    (epochAdmit verify ⟨cs, e₁⟩ m).network =
      (epochAdmit verify ⟨cs, e₂⟩ m).network

theorem epochAdmit_clock_monotone (verify : SigVerifier) (es : EpochNetwork)
    (m : GossipMessage) :
    es.network.gossip.latestUpdateTimestamp ≤
      (epochAdmit verify es m).network.gossip.latestUpdateTimestamp

theorem latestTimestamp_insertUpdateCanonical_ge {u : ChannelUpdate}
    {l : List ChannelUpdate} (h : UpdateSupersedes u l) :
    latestTimestamp l ≤ latestTimestamp (insertUpdateCanonical u l)

theorem stale_update_not_supersedes {u v : ChannelUpdate}
    {l : List ChannelUpdate} (hv : v ∈ l)
    (hscid : v.shortChannelId = u.shortChannelId)
    (hdir : v.direction = u.direction) (hts : u.timestamp ≤ v.timestamp) :
    ¬ UpdateSupersedes u l

theorem admitGossipCanonical_stale_update_eq (verify : SigVerifier)
    (cs : CanonicalNetwork) (u : ChannelUpdate)
    (hstale : ∃ v ∈ cs.gossip.updates,
      v.shortChannelId = u.shortChannelId ∧ v.direction = u.direction ∧
        u.timestamp ≤ v.timestamp) :
    admitGossipCanonical verify cs (.channelUpdate u) = cs

theorem canonical_superseded_update_stays_rejected (verify : SigVerifier)
    (cs : CanonicalNetwork) (u v : ChannelUpdate)
    (hgate : u.signatureValid verify ∧
      cs.gossip.updateKeyAnnouncedDirectional u ∧
        UpdateSupersedes u cs.gossip.updates)
    (hscid : v.shortChannelId = u.shortChannelId)
    (hdir : v.direction = u.direction) (hts : v.timestamp ≤ u.timestamp) :
    admitGossipCanonical verify
      (admitGossipCanonical verify cs (.channelUpdate u)) (.channelUpdate v) =
      admitGossipCanonical verify cs (.channelUpdate u)

theorem epoch_supersession_preserves_stale_rejection (verify : SigVerifier)
    (es : EpochNetwork) (u v : ChannelUpdate)
    (hgate : u.signatureValid verify ∧
      es.network.gossip.updateKeyAnnouncedDirectional u ∧
        UpdateSupersedes u es.network.gossip.updates)
    (hscid : v.shortChannelId = u.shortChannelId)
    (hdir : v.direction = u.direction) (hts : v.timestamp ≤ u.timestamp) :
    let es' := epochAdmit verify es (.channelUpdate u)
    (epochAdmit verify es' (.channelUpdate v)).network = es'.network

inductive ReachableEpoch (verify : SigVerifier) : EpochNetwork → Prop
  | empty (nodeKeyOf : NodeId → PubKey)
  | deliver : ReachableEpoch verify es →
      ReachableEpoch verify (epochAdmit verify es m)
  | advance : ReachableEpoch verify es →
      ReachableEpoch verify (epochAdvance es)
  | rotate (h : ReachableEpoch verify es)
      (hstore : StoreAnnouncementsValid verify (rekeyStore es.network.gossip n k))
      (hupdates : StoreUpdatesDirectionallyAttributed verify
        (rekeyStore es.network.gossip n k)) :
      ReachableEpoch verify (epochRotate es n k)

theorem rotateNetwork_preserves_canonicalAdmitted (verify : SigVerifier)
    (cs : CanonicalNetwork) (n : NodeId) (k : PubKey)
    (hinv : CanonicalAdmitted verify cs)
    (hstore : StoreAnnouncementsValid verify (rekeyStore cs.gossip n k))
    (hupdates : StoreUpdatesDirectionallyAttributed verify
      (rekeyStore cs.gossip n k)) :
    CanonicalAdmitted verify (rotateNetwork cs n k)

theorem rotateNetwork_preserves_canonicalAdmitted_of_noUpdates
    (verify : SigVerifier) (cs : CanonicalNetwork) (n : NodeId) (k : PubKey)
    (hinv : CanonicalAdmitted verify cs) (hupd : cs.gossip.updates = [])
    (hstore : StoreAnnouncementsValid verify (rekeyStore cs.gossip n k)) :
    CanonicalAdmitted verify (rotateNetwork cs n k)

theorem reachableEpoch_invariant {verify : SigVerifier} {es : EpochNetwork}
    (h : ReachableEpoch verify es) :
    CanonicalAdmitted verify es.network

theorem rotation_preserves_announcement_identity {cs : CanonicalNetwork}
    {n : NodeId} {k : PubKey} {a : ChannelAnnellation}
    (ha : a ∈ cs.gossip.announcements) :
    ∃ b ∈ (rekeyStore cs.gossip n k).announcements,
      b.shortChannelId = a.shortChannelId ∧ b.node1 = a.node1 ∧
        b.node2 = a.node2 ∧ b.capacitySat = a.capacitySat

theorem rotation_current_key_survives {cs : CanonicalNetwork}
    {n : NodeId} {k : PubKey} {a : ChannelAnnellation}
    (ha : a ∈ cs.gossip.announcements) (hn : a.node1 = n) :
    ∃ b ∈ (rekeyStore cs.gossip n k).announcements,
      b.node1 = n ∧ b.nodeKey1 = k

theorem stale_update_rejected_in_every_epoch :
    ∃ (verify : SigVerifier) (es : EpochNetwork) (u v : ChannelUpdate),
      ReachableEpoch verify es ∧ v ∈ es.network.gossip.updates ∧

    ∃ (verify : SigVerifier) (es : EpochNetwork) (n : NodeId) (k : PubKey),

theorem rotation_identity_witness :

## Scope caveats

- **The epoch counter is administrative, not a protocol clock.**  It is a
  `Nat` field on `EpochNetwork`, not a wire field and not wall-clock time.
  `epoch_admission_epoch_independent` states exactly that admission ignores it;
  `epochAdmit` is the unchanged `admitGossipCanonical` plus `epoch + 1`.
- **Two clocks only.**  Freshness remains (i) the per-slot timestamp gate
  `UpdateSupersedes` and (ii) the raw global `latestUpdateTimestamp`.
  `latestTimestamp_insertUpdateCanonical_ge` proves the canonical replacement
  never lowers the global clock; `epochAdmit_clock_monotone` lifts that to
  delivery; `epochAdvance_clock_eq`/`epochRotate_clock_eq` show epoch movement
  does not touch either clock.  The wave-7 boundary
  `canonical_admission_strictly_weaker_than_raw_clock` is re-exported unchanged:
  the raw global clock remains strictly stronger than the canonical per-slot
  gate.
- **A stale update is rejected by the gate, not by the epoch.**  If a stored
  record in the same `(short_channel_id, direction)` slot is at least as new as
  the candidate, `UpdateSupersedes` fails and the canonical step is a no-op; the
  concrete `stale_update_rejected_in_every_epoch` witness shows that iterating
  `epochAdmit` with any epoch value cannot make the stale update stick.
- **"Stale announcement" means pre-rotation key registration.**  After an
  explicit rotation the stored record carries the new key; an announcement
  still carrying the old key fails `GossipState.CompatibleWithStore` and is a
  no-op in every epoch (`stale_announcement_rejected_in_every_epoch`).  This is
  a registration rejection, not a timestamp or signature rejection.
- **Rotation premises are explicit and cannot be dropped.**
  `rotateNetwork_preserves_canonicalAdmitted` requires
  `StoreAnnouncementsValid verify (rekeyStore cs.gossip n k)` (announcements are
  re-signed) and `StoreUpdatesDirectionallyAttributed verify
  (rekeyStore cs.gossip n k)` (stored updates are re-signed/resubmitted under
  the rotated keys).  Without the second premise the wave-6 boundary
  `rotation_breaks_update_attribution` still applies and is re-exported here.
  With an empty update list only the re-signing premise remains
  (`..._of_noUpdates`).
- Rotation rewrites key fields only; `short_channel_id`, endpoints, capacity,
  node announcements, updates, and views are untouched.  It is not part of
  `admitGossipCanonical` and does not change raw `applyGossip`.
- No liveness, dissemination, chain-hash filtering, UTXO confirmation, or
  computational unforgeability is added.  `SigVerifier` remains abstract (C3).
- This module defines its own `EpochNetwork` and does not import `H39`/`H40`;
  no duplicate-name conflict exists because everything is namespaced under
  `LnLean.Proofs.WaveH.H41`.
